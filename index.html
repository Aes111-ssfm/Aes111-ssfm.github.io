Примеры кода

Python: Отправка запроса

import deepseek

client = deepseek.Client(api_key="your_api_key")
response = client.chat(
    model="deepseek-ultra",
    messages=[{"role": "user", "content": "Привет!"}]
)
print(response.choices[0].message.content)
